//3. Calcular el monto del credito, los meses totales, la tasa mensual, la cuota mensual y el ratio de endeudamiento.

//Definición de constantes simbólicas
INGRESO_MIN = 1500 ;
ENGANCHE_MIN = 0.20 ;
MAX_DEUDA = 0.40 ;
MESES_YEAR = 12 ;

//Lectura y de claracion de variables
Leer ingreso_mensual ;
Leer val_inmueble ;
Leer enganche ;
Leer plazo ;
Leer interes ;
Leer compromisos ;

//Calculos financieros iniciales
monto_credito = val_inmueble - enganche ;
meses_totales = plazo * MESES_YEAR ;
tasa_mensual = (interes / 100) / MESES_YEAR ;

//Calculo de la cuota mensual (Farmula de amortizacion estandar)
cuota_mensual = (monto_credito * tasa_mensual) / (1 - (1 + tasa_mensual) ^ (-meses_totales)) ;

//Calculo del ratio de endeudamiento total
ratio_endeudamiento = (cuota_mensual + compromisos) / ingreso_mensual ;

// Aplicar descuento del 5% al monto del credito final
monto_final = monto_credito * 0.95 ;



//Evaluacion de expresiones logicas compuestas para aprobar o rechazar
Si (ingreso_mensual >= INGRESO_MIN) Y (enganche >= (val_inmueble * ENGANCHE_MIN)) Y (ratio_endeudamiento <= MAX_DEUDA) entonces ;
estado_credito = "APROBADO" ;
Sino ; 
estado_credito = "RECHAZADO" ;
FinSi ;

//Mostrar los resultados por pantalla
Escribir "--- EVALUACION DE CREDITO HIPOTECARIO ---" ;
Escribir "Estado del Credito: " + estado_credito ;
Escribir "Monto del credito solicitado: " + monto_credito ;
Escribir "Monto del credito con 5% de descuento promocional: " + monto_final ;
Escribir "Cuota mensual estimada: " + cuota_mensual ;
Escribir "Ratio de endeudamiento: " + ratio_endeudamiento ;
Escribir "Compromisos financieros actuales:" + compromisos ;

// Seccion de conclusiones
Escribir "--------------------------------------------------" ;

Escribir "   de endeudamiento total no supere el limite maximo del 40%." ;

Escribir "   el riesgo financiero y la cuota mensual a pagar." ;
