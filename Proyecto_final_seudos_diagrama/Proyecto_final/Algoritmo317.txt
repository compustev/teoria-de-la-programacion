/* Se usa la constante simbolica TASA_CAMBIO para centralizar el valor del cambio de moneda,
lo que permite actualizar la tasa en una sola linea sin modificar la formula en el codigo. */
//Constantes
TASA_CAMBIO = 1.08 ;
//Lectura de variables
Leer euros ;
//Evaluacion de verificar 
Si (euros > 0) entonces ;
dolares = euros * TASA_CAMBIO ;
raiz_resultado = rc(abs(dolares)) ;
codigo_exito = 0 ;
Escribir euros + " euros equivalen a " + dolares + " dolares" ;
Escribir "La raiz cuadrada del valor absoluto es: ", raiz_resultado ;
Escribir "Codigo de finalizacion: ", codigo_exito ;
SiNo ;
Escribir "Error: El monto en euros debe ser mayor a cero." ;
