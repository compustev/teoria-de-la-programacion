//Descripción breve de los pasos a seguir:
//1. Leer los ingresos y los gastos familiares.
//2. Calcular la suma de gastos, el ahorro y el presupuesto base.
//3. Aplicar un descuento del 5% al resultado final.
//4. Evaluar si los gastos de vivienda y alimentación respetan los límites permitidos.
//5. Mostrar los resultados y alertas por pantalla.

//Definición de constantes simbolicas
PORCENTAJE_MAX_VIVIENDA = 0.30 ;
PORCENTAJE_MAX_ALIMENTACION = 0.20 ;
PORCENTAJE_MIN_AHORRO = 0.10 ;

//Lectura de datos y declaracion de variables 
leer ingreso_total ;
leer gasto_vivienda ;
leer gasto_alimentacion ;
leer gasto_transporte ;
leer gasto_educacion ;

//Calculos
suma_gastos = gasto_vivienda + gasto_alimentacion + gasto_transporte + gasto_educacion ;
ahorro = (ingreso_total - suma_gastos) ;
presupuesto = gasto_vivienda + gasto_alimentacion + gasto_transporte + gasto_educacion;   
presupuesto_des = presupuesto * 0.95 ;

//Evaluaciones y validaciones de limites con las constantes
Si gasto_vivienda > (ingreso_total * PORCENTAJE_MAX_VIVIENDA) entonces ;
Escribir "Gasto de vivienda se excede al presupuesto" ;
SiNo ;
Escribir "Gasto de vivienda  esta dentro del presupuesto" ;
FinSi ;
Si gasto_alimentacion > (ingreso_total * PORCENTAJE_MAX_ALIMENTACION) entonces ;
Escribir "Gasto de alimento se excede al presupuesto" ;
SiNo ;
Escribir "Gasto de alimento esta dentro del presupuesto" ;
FinSi ;
Si ahorro < (ingreso_total * PORCENTAJE_MIN_AHORRO) entonces ;
Escribir "El ahorro es inferior al 10% minimo requerido." ;
SiNo ;
Escribir "El ahorro cumple con la meta minima requerida." ;
FinSi ;
//Mostrar el presupuesto y resultados finales por pantalla
Escribir "--- PRESUPUESTO FAMILIAR ---" ;
Escribir "Presupuesto total (suma de gastos): " + presupuesto ;
Escribir "Presupuesto con el 5% de descuento: " + presupuesto_des ;
Escribir "Ahorro calculado: " + ahorro ;
Fin
